backend/
  .env                     # 后端环境配置文件
  .gitignore               # Git 忽略规则
  go.mod                   # Go 模块描述文件
  go.sum                   # Go 依赖校验文件
  main.go                  # 后端服务入口，启动 Gin 路由
  server.exe               # 编译后的可执行文件

  config/
    config.go              # 配置加载与环境变量处理

  database/
    database.go            # 数据库连接初始化与全局 DB 对象

  model/
    models.go              # GORM 数据模型定义：User、Team、Project、Recruitment、Response、Application 等

  middleware/
    auth.go                # 身份认证、权限检查中间件（RequireAuth、RequireRole）
    jwt.go                 # JWT Token 生成与解析

  handler/
    auto.go                # 用户注册、登录、当前用户信息、角色申请、管理员管理、超级管理员管理
    application.go         # 创业申请 CRUD 与审批流程
    team.go                # 团队创建、成员管理、邀请、入队申请、审核、转移所有权
    project.go             # 项目发布、上架/下架审核、作废、分页列表
    recruitment.go         # 团队招募发布、更新、解决、作废、分页列表
    response.go            # 对接响应创建、审批、作废
    resource.go            # 资源机会管理（partner/admin 权限）
    event.go               # 活动发布、更新、报名、管理员报名列表
    upload.go              # 文件上传处理

  repository/
    application_repository.go  # 申请相关数据库访问
    team_repository.go         # 团队、成员、邀请、入队申请数据库访问
    user_repository.go         # 用户查询与更新数据库访问

  service/
    auth_service.go         # 注册、登录、角色申请、管理员管理、超级管理员管理业务逻辑
    team_service.go         # 团队邀请、入队申请、审批与所有权转移业务逻辑

  response/
    response.go            # 统一响应格式封装

  router/
    router.go              # 路由注册与跨域配置
